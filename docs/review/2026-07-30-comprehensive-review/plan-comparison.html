    description string

    if v, ok := b.data["nodeConfig"].(NodeConfig); ok { return v }
    return NodeConfig{}
}
func (b *BaseSkill) SetNodeConfig(cfg NodeConfig) RunnableInterface {
    b.data["nodeConfig"] = cfg

ToMap/FromMap — trivial map copy (~15 lines total)
func (b *BaseSkill) ToMap() map[string]any {
    m := make(map[string]any, len(b.data)+2)
    for k, v := range b.data { m[k] = v }
    m["id"] = b.id
    m["description"] = b.description
    return m
}
func (b *BaseSkill) FromMap(m map[string]any) {
    b.data = make(map[string]any, len(m))
    for k, v := range m { b.data[k] = v }
    if v, ok := m["id"].(string); ok { b.id = v }
    if v, ok := m["description"].(string); ok { b.description = v }
}

BaseSkill — state lives in omni.Atom (thread-safe, serializable)
type BaseSkill struct {
    atom omni.AtomInterface  // ALL state here
}

func NewBaseSkill() *BaseSkill {
    return &BaseSkill{atom: omni.NewAtom("skill")}
}

func (b *BaseSkill) GetNodeConfig() NodeConfig {
    s := b.atom.Get("nodeConfig")  // RLock-protected
    return deserializeNodeConfig(s)
}
func (b *BaseSkill) SetNodeConfig(cfg NodeConfig) RunnableInterface {
    b.atom.Set("nodeConfig", serializeNodeConfig(cfg))  // Lock-protected
    return b
}
ToMap/FromMap — delegate to omni (~10 lines total)
func (b *BaseSkill) ToMap() map[string]any {
    return b.atom.ToMap()  // omni built-in
}
func (b *BaseSkill) FromMap(m map[string]any) {
    atom, _ := omni.NewAtomFromMap(m)  // omni built-in
    b.atom = atom
}
Skill body — UNCHANGED (getters read from Atom internally)
func (u *UserCreate) Run() types.Result {
    cfg := u.GetNodeConfig()        // reads Atom property
    username := u.GetArg("username") // reads Atom property
    if cfg.IsDryRunMode { ... }
}
Framework call — ToMap, replace, clone
m := skill.ToMap()               // omni.Atom.ToMap()
props := m["properties"].(map[string]string)
props["nodeConfig"] = serialize(n.cfg)
props["dryRun"] = "true"
clone := cloneFromMap(skill, m)  // fresh instance
result := clone.Run()

Pros & Cons Summary

1. Runnable Options Interface

Pros

  • Type-safe (interface methods, compile-time checked)
  • ctx as first param — contextcheck linter works
  • Future-proof (ctx in signature now)
  • No external dependency
  • Zero boilerplate
  • Interface is mockable in tests

Cons

  • Major breaking change (~80+ files)
  • Every skill author must rewrite Run/Check
  • Third-party skills break
  • Interface to implement/mock in tests
  • Limited third-party extensibility (fixed interface)
  • No serialization

2. Functional Options

Pros

  • Type-safe (struct fields, compile-time checked)
  • Simple for skill authors (read struct fields directly)
  • No interface to implement/mock
  • Adding RunConfig fields is non-breaking for readers
  • No external dependency
  • Zero boilerplate

Cons

  • Major breaking change (~80+ files)
  • contextcheck linter broken (ctx is an option)
  • Every skill author must rewrite Run/Check
  • Third-party skills break
  • Limited third-party extensibility (fixed struct)
  • No serialization

3. Deep Copy

Pros

  • Non-breaking — zero API change
  • Only ~3 files changed
  • Zero skill author impact
  • Zero third-party impact (automatic)
  • Zero boilerplate
  • Handles extra skill fields automatically
  • Fastest migration (hours)

Cons

  • External dependency (brunoga/deep)
  • Uses unsafe for unexported fields
  • No ctx future-proofing (needs break later)
  • ~10µs reflection overhead per call (negligible vs SSH)
  • Non-copyable skill fields (channels, funcs) need custom Copier
  • No serialization

4. JSON Serialize

Pros

  • Non-breaking — zero API change
  • Only ~5 files changed
  • Zero skill author impact
  • Zero third-party impact (automatic)
  • No external dependency (stdlib only)
  • Serialized form reusable (logging, auditing)

Cons

  • Silent data loss risk — proxy out of sync with BaseSkill
  • ~50 lines custom boilerplate (MarshalJSON)
  • Extra skill fields need custom MarshalJSON
  • ~50-200µs JSON overhead per call
  • No ctx future-proofing
  • Uses reflect for type discovery

5. Map-Backed Clone

Pros

  • Skill bodies unchanged — Run()/Check() stay as-is
  • Only ~4 files changed
  • Free serialization (json.Marshal(ToMap()))
  • No external dependency (stdlib only)
  • No unsafe (uses existing getters/setters)
  • Fast (~2-5µs per clone)
  • Enables playbooks, remote workers, audit logging
  • Automatic for skills embedding *BaseSkill
  • Follows dracory/ui Block pattern

Cons

  • Adds 2 methods to interface (minor break for non-embedding implementors)
  • ~50 lines boilerplate (ToMap/FromMap on BaseSkill + commandImpl)
  • Silent data loss risk if BaseSkill gains field not in ToMap
  • Extra skill fields need ToMap/FromMap override
  • Uses reflect in cloneSkill (safe ops, but reflect)
  • No ctx future-proofing (needs break later)

6. Map-Storage Runnable

Pros

  • No silent data loss — new field = new map key, automatic
  • Trivial ToMap/FromMap — ~15 lines (map copy, no conversion)
  • No per-type overrides (commandImpl fields are just map keys)
  • Skill bodies unchanged — getters read from map internally
  • Only ~4 files changed
  • Free serialization (json.Marshal(ToMap()))
  • No external dependency, no unsafe
  • Fast (~2-5µs per clone)
  • True dracory/ui Block pattern — map IS the storage
  • Store typed values directly (no string conversion)

Cons

  • BaseSkill internally restructured (typed fields → map)
  • Adds 2 methods to interface (minor break for non-embedding implementors)
  • Type assertions in getters (comma-ok, no panic, but not compile-time)
  • Uses reflect in cloneFromMap (safe ops, but reflect)
  • Shallow copy in ToMap — nested maps shared (framework replaces, doesn't mutate)
  • No ctx future-proofing (needs break later)

7. Omni-Backed Runnable

Pros

  • Thread-safe by default — sync.RWMutex in every Atom
  • Zero serialization code — ToMap/ToJSON/ToGob built-in
  • Defense-in-depth — clone + mutex (double protection)
  • No silent data loss (all state in Atom, automatic)
  • No per-type overrides (commandImpl fields are Atom properties)
  • Skill bodies unchanged — getters delegate to Atom
  • JSON + Gob + Map serialization (all built-in)
  • Children/hierarchy support (future: skill groups)
  • Functional options (WithID, WithProperties)
  • Memory profiling (atom.MemoryUsage())
  • Tested library with benchmarks
  • Zero transitive dependencies

Cons

  • External dependency (dracory/omni)
  • Atom stores map[string]string — NodeConfig needs JSON string serialization
  • BaseSkill internally restructured (typed fields → Atom)
  • Adds 2 methods to interface (minor break for non-embedding implementors)
  • Uses reflect in cloneFromMap (safe ops, but reflect)
  • Mutex overhead on every getter/setter (~25ns, negligible)
  • No ctx future-proofing (needs break later)

Safety Mechanism Comparison

Plan What eliminates the race Shared state after fix
1. Opts Interface Framework builds fresh RunnableOptionsInterface per call; skill has no execution setters Skill instance is read-only (only id/description)
2. Functional Opts Skill applies opts to a local RunConfig on the stack; skill has no execution setters Skill instance is read-only (only id/description)
3. Deep Copy Framework clones skill before mutating; runs the clone, not the original Original skill instance is never mutated; each goroutine has its own clone
4. JSON Serialize Framework serializes skill, deserializes into fresh instance; runs the fresh instance Original skill instance is never mutated; each goroutine has its own deserialized copy
5. Map Clone Framework clones skill via FromMap(ToMap()) before mutating; runs the clone Original skill instance is never mutated; each goroutine has its own clone
6. Map Storage Framework does ToMap → modify map → cloneFromMap; runs the fresh instance Original skill instance is never mutated; each goroutine has its own clone with its own map
7. Omni-Backed Framework does ToMap → modify → cloneFromMap; runs the fresh instance. Atom's RWMutex adds defense-in-depth Original skill never mutated; each goroutine has its own clone. Even if accessed concurrently, Atom's mutex prevents fatal panics
